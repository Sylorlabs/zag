set pagination off
set confirm off
break *0x4c08df
commands
  silent
  x/8i $pc-24
  printf "RDI=%p RAX=%p R12=%p RET=", $rdi, $rax, $r12
  x/gx $rsp
  printf "FREE_READ_CALL descriptor words: "
  x/2gx $r12
  continue
end
catch syscall write
commands
  silent
  if ($rdi == 2 && $rdx == 51)
    printf "PANIC_WRITE rip=%p rsi=%p rdx=%ld\n", $rip, $rsi, $rdx
    x/s $rsi
    x/i $rip
    bt 8
    quit
  end
  continue
end
run
